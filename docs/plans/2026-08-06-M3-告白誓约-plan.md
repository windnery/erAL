# M3 告白/誓约 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 relationship 陷落阶段推进事件（友好→喜欢→爱）、告白与誓约指令，以及誓约后的特权。

**Architecture:** 新增 `EventManager`（`game_engine/events/relationship.py`），`CommandManager.do_cmd` 在目标指令成功执行后调用 `world.event_manager.check_npc_events(npc)`，favor/trust 跨过阈值时推进 `talent['relationship']` 并返回事件文案（只升不降、同阶段只触发一次）。`confess`（告白：relationship=爱 且好感/信赖达标，90% 成功率）与 `marry`（誓约：需消耗背包中的"誓约之戒"）注册为约会类指令。誓约特权：`cflag['married']` 每日自动涨好感、hug 门槛减半、前端状态标记。所有新状态（lover/married）都在 cflag 中，自动纳入 M0 存档协议。

**Tech Stack:** Python 3.11+ 标准库（unittest），前端原生 JS（cflag 标记自动渲染）。

`game_engine/events/relationship.py`：

```python
from config.attr_defs import ATTR_DEFS
from config.relationship import RELATIONSHIP_THRESHOLDS
from game_engine.models.shipgirl import ShipGirl


def check_relationship_events(npc: ShipGirl) -> list[str]:
    """favor/trust 跨过阈值时推进 relationship，返回事件文案列表（可能为空）"""
    current = npc.get_talent_value('relationship')
    if current >= 3:
        return []
    target = current + 1
    threshold = RELATIONSHIP_THRESHOLDS[target]
    if npc.favor < threshold['favor'] or npc.trust < threshold['trust']:
        return []

    npc.talent['relationship'] = str(target)
    texts = []
    line = npc.get_line(f'relationship_event_{target}')
    if line:
        texts.append(line.replace('{name}', npc.name))
    stage_name = ATTR_DEFS['talent']['relationship']['value'][str(target)]
    texts.append(f'和{npc.name}的关系变得「{stage_name}」了！')
    return texts
```

from game_engine.events.relationship import check_relationship_events


class EventManager:
    """世界事件检查器：指令结算后被 do_cmd 调用"""

    def __init__(self, world):
        self.world = world

    def check_npc_events(self, npc) -> list[str]:
        """检查目标舰娘的事件（目前仅 relationship 推进）"""
        return check_relationship_events(npc)

```python
MARRY_MIN_FAVOR = 10000
CONFESS_MIN_FAVOR = 5000
CONFESS_MIN_TRUST = 2000


def can_confess(world: World, npc: ShipGirl):
    '''告白判定：关系到达「爱」'''
    if not global_can(world.player, npc):
        return False
    if npc.get_talent_value('relationship') < 3:
        return False
    if npc.cflag.get('lover', False) or npc.cflag.get('married', False):
        return False
    if npc.favor < CONFESS_MIN_FAVOR or npc.trust < CONFESS_MIN_TRUST:
        return False
    return True


def can_marry(world: World, npc: ShipGirl):
    '''誓约判定：关系到达「爱」且持有誓约之戒'''
    if not global_can(world.player, npc):
        return False
    if npc.get_talent_value('relationship') < 3:
        return False
    if npc.cflag.get('married', False):
        return False
    if npc.favor < MARRY_MIN_FAVOR:
        return False
    if 'ring' not in world.player.inventory:
        return False
    return True


@register_cmd('confess', '告白', '约会', can_confess)
def confess(world: World, option: str):
    '''向舰娘告白'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)

    ctx.say(f'鼓足勇气，向{npc.name}告白了……')
    accepted = randint(1, 100) <= 90
    if not accepted:
        line = npc.get_line('confess_fail')
        if line:
            ctx.say(line.replace('{name}', npc.name))
        npc.favor -= 500
        ctx.say(f'好感-500 ({npc.name})')
        ctx.say('告白被婉拒了……')
        return ctx.result()

    npc.cflag['lover'] = True
    line = npc.get_line('confess_success')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'告白成功！{npc.name}成为了你的恋人。')
    return ctx.result()


@register_cmd('marry', '誓约', '约会', can_marry)
def marry(world: World, option: str):
    '''与舰娘誓约（消耗誓约之戒）'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)

    world.player.inventory.remove('ring')
    npc.talent['relationship'] = '4'
    npc.cflag['married'] = True
    npc.cflag['lover'] = True

    ctx.say(f'将誓约之戒轻轻戴在了{npc.name}的手上……')
    line = npc.get_line('marry')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'誓约成立！{npc.name}成为了你的伴侣。')
    return ctx.result()
```
