# M1 约会系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现完整的约会流程：邀请（判定成功/失败）→ 同行 → 约会地点互动 → 手动/跨天强制结束，并扩展口上条件系统。

**Architecture:** 新增 `game_engine/commands/date.py` 注册三个指令（`date_invite`/`date_action`/`end_date`），约会状态用已有 `cflag['dating']`（随 cflag 自动纳入 M0 存档协议，无需改 SaveManager）。`NpcManager.update_positions` 中约会中的舰娘跳过睡觉逻辑并始终跟随玩家；跨天强制结束逻辑由 `NpcManager.end_date_if_needed()` 集中实现，`update_positions` 与 `settle_day` 两处调用（后者拼接文案）。口上条件在 `ShipGirl._check_conditions` 扩展 `min_relationship`/`max_relationship`/`is_dating` 三类。

**Tech Stack:** Python 3.11+ 标准库（unittest），前端原生 JS（选择幕复用）。

`game_engine/commands/date.py`：

```python
from __future__ import annotations
from typing import TYPE_CHECKING
from random import randint

from config.attr_defs import ATTR_DEFS
from game_engine.commands._commands import register_cmd
from game_engine.commands._common import new_source, global_can
from game_engine.commands._context import CommandContext
from game_engine.data_pipeline.common_src_modify import common_src_modify
from game_engine.data_pipeline.favor.favor_calc import favor_calc
from game_engine.data_pipeline.palam.palam_calc import palam_calc
from game_engine.data_pipeline.trust.trust_calc import trust_calc
from game_engine.models.shipgirl import ShipGirl
from data.time.time_data import command_time_data

if TYPE_CHECKING:
    from world import World

DATE_INVITE_MIN_FAVOR = 1000
DATE_INVITE_MIN_RELATIONSHIP = 1


def can_invite(world: World, npc: ShipGirl):
    '''约会邀请判定'''
    if not global_can(world.player, npc):
        return False
    if npc.is_dating():
        return False
    if npc.get_talent_value('relationship') < DATE_INVITE_MIN_RELATIONSHIP:
        return False
    if npc.favor < DATE_INVITE_MIN_FAVOR:
        return False
    return True


def can_date_action(world: World, npc: ShipGirl):
    '''约会行动判定'''
    if not global_can(world.player, npc):
        return False
    if not npc.is_dating():
        return False
    return True


def can_end_date(world: World, npc: ShipGirl):
    '''结束约会判定'''
    if not npc.is_dating():
        return False
    return True


@register_cmd('date_invite', '约会', '约会', can_invite)
def date_invite(world: World, option: str):
    '''邀请舰娘约会'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)

    # 邀请成功率：基础 50% + 好感加成（每 1000 好感 +5%，上限 95%）
    success_rate = min(50 + npc.favor // 1000 * 5, 95)
    accepted = randint(1, 100) <= success_rate

    if not accepted:
        line = npc.get_line('date_invite_refused')
        ctx.say(f'向{npc.name}发出了约会邀请……')
        if line:
            ctx.say(line.replace('{name}', npc.name))
        npc.favor -= 50
        ctx.say(f'好感-50 ({npc.name})')
        ctx.say('她委婉地拒绝了。')
        return ctx.result()

    npc.cflag['dating'] = True
    npc.cflag['following'] = True
    line = npc.get_line('date_invite')
    ctx.say(f'向{npc.name}发出了约会邀请……')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'约会开始！和{npc.name}一起行动吧。')
    return ctx.result()


@register_cmd('date_action', '约会行动', '约会', can_date_action)
def date_action(world: World, option: str):
    '''约会中的地点互动：option 为目标节点 key'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)
    target_node = world.player.location['node']

    node_name = world.map_manager.maps[world.player.location['region']][target_node]['name']
    ctx.say(f'在{node_name}和{npc.name}一起散步……')

    line = npc.get_line('date_action')
    if line:
        ctx.say(line.replace('{name}', npc.name))

    source: dict[str, int] = new_source({
        'happiness_source': 300,
        'love_source': 150,
        'conquest_source': 50,
    })
    source = common_src_modify(source, npc)

    source_list = []
    for k, v in source.items():
        if v != 0:
            source_list.append(f"{ATTR_DEFS['source'][k]['name']}({v})")
    ctx.say(' '.join(source_list))

    mes_source, mes_target = palam_calc(source, world.player, npc)
    for mes in mes_source:
        ctx.say(mes)
    for mes in mes_target:
        ctx.say(mes)
    world.player.update_palam_level()
    npc.update_palam_level()

    favor_delta = favor_calc(npc, source)
    trust_delta = trust_calc(npc, source)
    npc.favor += favor_delta
    npc.trust += trust_delta

    if favor_delta > 0:
        ctx.say(f'好感+{favor_delta} ({npc.name})')
    if trust_delta > 0:
        ctx.say(f'信赖+{trust_delta} ({npc.name})')

    ctx.advance_time(command_time_data['date_action'])
    ctx.say(f'度过了{command_time_data["date_action"]}分钟')
    return ctx.result()


@register_cmd('end_date', '结束约会', '约会', can_end_date)
def end_date(world: World, option: str):
    '''结束约会'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)

    npc.cflag['dating'] = False
    npc.cflag['following'] = False
    # 心情结算：约会顺利则心情上升
    mood_delta = randint(1, 3)
    npc.base['mood'] = npc.base.get('mood', 0) + mood_delta

    line = npc.get_line('end_date')
    ctx.say(f'和{npc.name}结束了今天的约会。')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'{npc.name}的心情似乎变好了（心情+{mood_delta}）')
    return ctx.result()
```

    DATE_END_START_HOUR = 22  # 深夜 22 点后强制结束约会

    def end_date_if_needed(self, sg: ShipGirl, hour: int, minutes: int) -> bool:
        """约会中且进入深夜 → 强制结束约会。返回是否发生了强制结束"""
        if not sg.cflag.get('dating', False):
            return False
        if hour >= self.DATE_END_START_HOUR or hour < 4:
            sg.cflag['dating'] = False
            sg.cflag['following'] = False
            sleep_loc = self.shipgirls_db[sg.id].get('location', {})
            if sleep_loc:
                sg.location = dict(sleep_loc)
            return True
        return False
