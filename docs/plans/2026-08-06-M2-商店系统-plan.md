# M2 商店系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增商店区域与两名驻点商人舰娘（明石卖皮肤、不知火卖日用品），实现购买/送礼/问候指令与玩家背包。

**Architecture:** 数据驱动的商品表 `data/shop/goods.json`（以店主舰娘 id 为 key），`ShopManager` 负责读取。明石/不知火通过 ShipGirl 新增 `station` 字段常驻商店（`NpcManager.update_positions` 驻点分支优先级最高）。购买指令 `buy`、问候指令 `shop_greet` 绑定店主 NPC；送礼指令 `give_gift` 消耗玩家 `inventory`（Player 新增字段并登记进 M0 存档协议）。`do_cmd` 扩展复合 option（`{npc_id, payload}`）以支持"目标 NPC + 商品/礼物选择"，前端 `makeCmdSpan` 同步适配（向后兼容，老指令无感知）。明石的商品表 M2 留空、M4 填充皮肤。

**Tech Stack:** Python 3.11+ 标准库（unittest），前端原生 JS（选择幕复用）。

```python
import json
from pathlib import Path

DATA_DIR = Path(__file__).parent.parent.parent / 'data'


class ShopManager:
    """商品表管理：data/shop/goods.json，以店主舰娘 id 为 key"""

    def __init__(self, goods_file: Path | None = None):
        path = goods_file or (DATA_DIR / 'shop' / 'goods.json')
        with open(path, 'r', encoding='utf-8') as f:
            self.goods_db: dict = json.load(f)

    def get_shop_goods(self, sg_id: str) -> list[dict]:
        """某店主店铺的全部商品"""
        return list(self.goods_db.get(sg_id, {}).get('goods', []))

    def get_goods(self, sg_id: str, goods_id: str) -> dict | None:
        """按店主+商品 id 查询"""
        for g in self.get_shop_goods(sg_id):
            if g['id'] == goods_id:
                return g
        return None

    def get_goods_anywhere(self, goods_id: str) -> dict | None:
        """跨店按商品 id 查询（背包道具名称显示用）"""
        for shop in self.goods_db.values():
            for g in shop.get('goods', []):
                if g['id'] == goods_id:
                    return g
        return None
```

```python
from __future__ import annotations
from typing import TYPE_CHECKING
from random import randint

from game_engine.commands._commands import register_cmd
from game_engine.commands._common import global_can
from game_engine.commands._context import CommandContext

if TYPE_CHECKING:
    from world import World

SHOPKEEPERS = {'akashi', 'shiranui'}


def _unpack(option) -> tuple[str, dict | None] | None:
    """从复合 option 拆出 npc_id 与 payload；非复合返回 None"""
    if isinstance(option, dict) and 'npc_id' in option:
        return option['npc_id'], option.get('payload')
    return None


def can_buy(world: World, npc):
    '''购买判定：目标是店主'''
    if not global_can(world.player, npc):
        return False
    return npc.id in SHOPKEEPERS


def can_greet(world: World, npc):
    '''问候判定：目标是店主'''
    if not global_can(world.player, npc):
        return False
    return npc.id in SHOPKEEPERS


@register_cmd('buy', '买东西', '日常', can_buy)
def buy(world: World, option):
    '''从店主处购买商品'''
    ctx = CommandContext(world)
    packed = _unpack(option)
    if not packed:
        return []
    npc_id, payload = packed
    if not payload:
        return []
    npc = world.npc_manager.get_npc_by_id(npc_id)
    goods_id = payload['goods_id']
    goods = world.shop_manager.get_goods(npc_id, goods_id)
    if goods is None:
        return ['没有这件商品']
    if world.player.money < goods['price']:
        return ['资金不足']

    world.player.set_money(world.player.money - goods['price'])

    gtype = goods.get('type', 'consumable')
    mes = f'购买了{goods["name"]}（{goods["price"]}资金）'
    if gtype == 'gift':
        world.player.inventory.append(goods_id)
        mes += f'，放入背包'
    elif gtype == 'skin':
        world.player.skins.setdefault(npc_id, []).append(goods_id)
        mes += f'，{npc.name}收下了新衣服'
    ctx.say(mes)

    line = npc.get_line('shop_buy')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'剩余资金：{world.player.money}')
    return ctx.result()


@register_cmd('shop_greet', '打招呼', '日常', can_greet)
def shop_greet(world: World, option: str):
    '''和店主打招呼'''
    ctx = CommandContext(world)
    npc = world.npc_manager.get_npc_by_id(option)
    line = npc.get_line('shop_greet')
    ctx.say(f'和{npc.name}打了个招呼……')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    return ctx.result()
```

```python
def can_give_gift(world: World, npc):
    '''送礼判定：背包有礼物'''
    if not global_can(world.player, npc):
        return False
    return len(world.player.inventory) > 0


def _gift_favor_delta(price: int) -> int:
    """按价格分档的好感增量"""
    if price < 200:
        return randint(5, 10)
    elif price <= 800:
        return randint(15, 25)
    else:
        return randint(40, 60)


@register_cmd('give_gift', '送礼', '亲昵', can_give_gift)
def give_gift(world: World, option):
    '''送礼物给舰娘'''
    ctx = CommandContext(world)
    packed = _unpack(option)
    if not packed:
        return []
    npc_id, payload = packed
    if not payload:
        return []
    npc = world.npc_manager.get_npc_by_id(npc_id)
    goods_id = payload['goods_id']
    if goods_id not in world.player.inventory:
        return []

    goods = world.shop_manager.get_goods_anywhere(goods_id)
    price = goods['price'] if goods else 0
    world.player.inventory.remove(goods_id)

    favor_delta = _gift_favor_delta(price)
    npc.favor += favor_delta
    npc.trust += max(1, favor_delta // 2)

    line = npc.get_line('gift')
    ctx.say(f'把{goods["name"] if goods else "礼物"}送给了{npc.name}……')
    if line:
        ctx.say(line.replace('{name}', npc.name))
    ctx.say(f'好感+{favor_delta} 信赖+{max(1, favor_delta // 2)} ({npc.name})')
    return ctx.result()
```
